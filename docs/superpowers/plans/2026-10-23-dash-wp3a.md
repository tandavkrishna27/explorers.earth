# WP3A: Category page template, Top Picks, New list popup — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** All 8 dashboard category pages (Places, Movies & Shows, Books, Games, Apps & Tools, Products, People, Guides) render from one `CategoryPage` template with an always-visible Top Picks hero, one shared Top Picks manager, list cards that match CAT-06/CAT-08 with drag-and-drop reorder (GL-06), and one shared New list popup (NL-01..05) that lands on the new list.

**Architecture:** New presentational template under `explorers-earth/src/features/shared/category/` fed by thin per-category wrappers (the existing `*Home.tsx`, the category part of `GuidesPage.tsx`, and step 1 of `pages/Favorites.tsx`). Wrappers keep their existing owner-read hooks and map their legacy list shapes to two small view models (`ListCardModel`, `TopPickModel`). All writes go through category-agnostic helpers that call `lib/explorersApiClient.ts` and `lib/reorderCategoryLists.ts` directly (no edits to any `features/*/api/*` file). Item Top Picks (six categories) use the category top-picks command; list Top Picks (Places, Guides) use the list `pinOrder` + pinned-scope order command. WP2 controls are consumed through a local stub with WP2's agreed props, swapped for a re-export when WP2 merges. No migration, no server code change (see "Verified against main"); the server side is proved with new cases in an already-registered integration file.

**Tech Stack:** React 18 + TypeScript (strict), Tailwind, framer-motion (already a dependency; used only for the manager sheet), react-i18next (`dashboard.wp3a.*` keys), Vitest + Testing Library (jsdom) through `scripts/run-contained-vitest.cjs`, Playwright (general e2e lane), tunes Vitest integration against PostgreSQL 15.

**Approved mocks (visual spec):** v3 dashboard https://claude.ai/artifact/D89UZnowRi1R9veMbYeWtb (category page, Top Picks hero, empty hero, list cards, dashed add card), v5 controls https://claude.ai/artifact/UDhwrVqLBNh8xKtZpppf6v (header: category name + publish pill top right + desktop "＋ New list"; dashed "＋ New list" FIRST card; grip "⋮⋮" on cards; phone bottom "＋ New list" button). Both were read for this plan.

```ts
export type ListStatus = "public" | "draft";

export interface ListCardModel {
  id: string;
  title: string;
  status: ListStatus;
  itemCount: number;
  pinnedCount: number;
  covers: string[];
}

export interface TopPickModel {
  id: string;
  title: string;
  subtitle: string | null;
  imageUrl: string | null;
  listId: string;
  listTitle: string;
  pinned: boolean;
  /** Lists mode only: a draft list cannot be pinned. */
  pinnable: boolean;
}

type LegacyItem = Record<string, any> & { documentId: string; is_pinned?: boolean; pin_order?: number | null };
type LegacyList = Record<string, any> & { documentId: string; List_Name?: string };

const unique = (items: LegacyItem[]) => {
  const seen = new Set<string>();
  return items.filter((item) => (seen.has(item.documentId) ? false : (seen.add(item.documentId), true)));
};

export function toListCards(lists: LegacyList[], spec: { relation: string; visibility: "Visibility" | "visibility"; cover: (item: LegacyItem) => string | null }): ListCardModel[] {
  return lists.map((list) => {
    const items = unique((list[spec.relation] as LegacyItem[] | undefined) ?? []);
    return {
      id: list.documentId,
      title: list.List_Name ?? "",
      status: list[spec.visibility] ? "public" : "draft",
      itemCount: items.length,
      pinnedCount: items.filter((item) => item.is_pinned).length,
      covers: items.map(spec.cover).filter((url): url is string => Boolean(url)).slice(0, 4),
    };
  });
}

export function toItemTopPicks(
  lists: LegacyList[],
  spec: { relation: string; title: (item: LegacyItem) => string; image: (item: LegacyItem) => string | null; subtitle: (item: LegacyItem) => string | null },
  pinnedOnly = true,
): TopPickModel[] {
  const seen = new Map<string, TopPickModel & { order: number }>();
  for (const list of lists) {
    for (const item of (list[spec.relation] as LegacyItem[] | undefined) ?? []) {
      if (seen.has(item.documentId) || (pinnedOnly && !item.is_pinned)) continue;
      seen.set(item.documentId, {
        id: item.documentId, title: spec.title(item), subtitle: spec.subtitle(item), imageUrl: spec.image(item),
        listId: list.documentId, listTitle: list.List_Name ?? "", pinned: Boolean(item.is_pinned), pinnable: true,
        order: item.pin_order ?? 999,
      });
    }
  }
  const picks = [...seen.values()];
  if (pinnedOnly) picks.sort((a, b) => a.order - b.order);
  return picks.map(({ order: _order, ...pick }) => pick);
}

/** Places and Guides: the lists themselves are the picks. */
export function toListTopPicks(lists: (LegacyList & { is_pinned?: boolean; pin_order?: number | null })[], spec: { title: (list: LegacyList) => string; image: (list: LegacyList) => string | null; isPublic: (list: LegacyList) => boolean }, pinnedOnly = true): TopPickModel[] {
  return lists
    .filter((list) => !pinnedOnly || list.is_pinned)
    .sort((a, b) => (pinnedOnly ? (a.pin_order ?? 999) - (b.pin_order ?? 999) : 0))
    .map((list) => ({ id: list.documentId, title: spec.title(list), subtitle: null, imageUrl: spec.image(list), listId: list.documentId,
      listTitle: spec.title(list), pinned: Boolean(list.is_pinned), pinnable: spec.isPublic(list) }));
}
```

```ts
import { useCallback, useEffect, useRef, useState } from "react";

export function moveCard<T>(items: readonly T[], from: number, to: number): T[] {
  const next = [...items];
  if (from === to || from < 0 || to < 0 || from >= next.length || to >= next.length) return next;
  const [moved] = next.splice(from, 1);
  next.splice(to, 0, moved);
  return next;
}

/** GL-06 for list cards: drag is primary; Space/arrows/Space and the ⋯ menu are the keyboard and touch paths. */
export function useCardReorder(savedOrder: string[], commit: (order: string[]) => Promise<unknown> | void) {
  const [order, setOrder] = useState(savedOrder);
  const [picked, setPicked] = useState<string | null>(null);
  const [dragging, setDragging] = useState<string | null>(null);
  const [over, setOver] = useState<string | null>(null);
  const before = useRef<string[]>(savedOrder);
  const savedKey = savedOrder.join("|");
  useEffect(() => { if (!picked && !dragging) setOrder(savedOrder); }, [savedKey]); // eslint-disable-line react-hooks/exhaustive-deps

  const save = useCallback((next: string[], revertTo: string[]) => {
    if (next.join("|") === revertTo.join("|")) return;
    setOrder(next);
    Promise.resolve(commit(next)).catch(() => setOrder(revertTo));
  }, [commit]);

  const onGripKey = (id: string, event: Pick<KeyboardEvent, "key" | "preventDefault">) => {
    const index = order.indexOf(id);
    if (event.key === " " || event.key === "Enter") {
      event.preventDefault();
      if (picked === id) { setPicked(null); save(order, before.current); }
      else { before.current = order; setPicked(id); }
    } else if (picked === id && (event.key === "ArrowRight" || event.key === "ArrowDown" || event.key === "ArrowLeft" || event.key === "ArrowUp")) {
      event.preventDefault();
      const delta = event.key === "ArrowRight" || event.key === "ArrowDown" ? 1 : -1;
      setOrder(moveCard(order, index, index + delta));
    } else if (picked === id && event.key === "Escape") {
      event.preventDefault(); setPicked(null); setOrder(before.current);
    }
  };
  const dragStart = (id: string) => { before.current = order; setDragging(id); };
  const dragOver = (id: string) => {
    if (!dragging || id === dragging) return;
    setOver(id);
    setOrder((current) => moveCard(current, current.indexOf(dragging), current.indexOf(id)));
  };
  const drop = () => { const next = order; setDragging(null); setOver(null); save(next, before.current); };
  const dragEnd = () => { if (dragging) { setDragging(null); setOver(null); setOrder(before.current); } };
  const moveBy = (id: string, delta: -1 | 1) => {
    const index = order.indexOf(id);
    save(moveCard(order, index, index + delta), order);
  };
  return { order, picked, dragging, over, onGripKey, dragStart, dragOver, drop, dragEnd, moveBy };
}
```

Note for `drop`: `drop()` reads `order` after `dragOver` updates; in the test each `act` flushes state, so the closure sees the moved order.

```ts
import { explorersApiClient } from "../../../lib/explorersApiClient";
import { reorderCategoryLists } from "../../../lib/reorderCategoryLists";
import { setListPin } from "../category/listWrites";
import type { ItemTopPickCategory } from "../category/categoryConfig";

export async function saveItemPins(category: ItemTopPickCategory, orderedRecommendationIds: string[]) {
  const observed = await explorersApiClient.getCompleteMyCategoryContent({ category, status: "active" }, undefined, true);
  const pins = orderedRecommendationIds.map((recommendationId) => {
    const saved = observed.topPicks?.find((pin) => pin.recommendationId === recommendationId);
    const live = observed.memberships.filter((member) => member.recommendationId === recommendationId && !member.collectionArchived && !member.recommendationArchived);
    const member = live.find((value) => value.collectionId === saved?.collectionId) ?? live[0];
    if (!member) throw new Error("This item changed. Refresh and try again.");
    return { recommendationId, collectionId: member.collectionId };
  });
  return explorersApiClient.setMyCategoryTopPicks(observed, pins, crypto.randomUUID());
}

/** Places and Guides: a pinned list carries pinOrder; the pinned scope then fixes the order densely. */
export async function saveListPins(category: "places" | "guides", previous: string[], next: string[]) {
  for (const id of previous) if (!next.includes(id)) await setListPin(id, null);
  const kept = next.filter((id) => previous.includes(id)).length;
  let position = kept;
  for (const id of next) if (!previous.includes(id)) await setListPin(id, position++);
  if (next.length > 0) await reorderCategoryLists(category, "pinned", next);
}
```

```tsx
  value.toLowerCase().replace(/[^a-z0-9\s-]/g, "").trim().replace(/\s+/g, "-").replace(/-+/g, "-").replace(/^-|-$/g, "").slice(0, 100);

const cityOf = (place: Places) => {
  for (const type of ["locality", "sublocality", "administrative_area_level_2", "administrative_area_level_3"]) {
    const component = place.address_components?.find((part) => part.types.includes(type));
    if (component) return component.long_name;
  }
  return place.name ?? "";
};
```
