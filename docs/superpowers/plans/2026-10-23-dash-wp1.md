# WP1: Shell, origin back, loaders, toasts — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Back always returns to the origin (GL-03), one loader owner per screen with the globe splash only at app boot reading "Getting your recommendations ready…" (GL-02, OWN-08 loader part), `sonner` as the only toast system at bottom centre (GL-05), and an addressable, test-locked route table for the seven non-Places categories (SCR-03).

**Architecture:** A small pure module `lib/navigation/originBack.ts` decides where Back goes (router `state.from` → validated `?redirectBack=` → caller fallback). A hook `useOriginBack` and a component `BackToOrigin` wrap it for pages. A `BootSplash` component is the single boot-time globe; `RouteLoader` (a 600 ms globe on every route change) is deleted so route changes fall back to `RouteFallback` / page skeletons. `PublicColdEntryBoundary` exposes `overlayVisible`, and `TabVisibilityGuard` renders nothing while it is true. Dead toast modules and the dead `Loader.tsx` are deleted.

**Tech Stack:** React 18, TypeScript strict, react-router-dom 7.15 (`BrowserRouter`, history state `{usr,key,idx}` survives reload), react-i18next 15, sonner 1.7, Vitest 4 via `scripts/run-contained-vitest.cjs`, Playwright 1.61.

**Base:** branch from current `main` (reference: 8063651c). All line numbers below are on 8063651c (read-only reference checkout `.claude/worktrees/wave1-main-ref/`).

**Mocks used:** v3 (D89UZnowRi1R9veMbYeWtb) and v5 (UDhwrVqLBNh8xKtZpppf6v) were read with the Artifact tool. Strings and sizes copied from them:

```ts
// explorers-earth/src/lib/navigation/originBack.ts
/**
 * Back always returns to the origin (GL-03).
 * Order: router state.from (in-app navigation; survives a browser refresh because
 * BrowserRouter keeps it in window.history.state) -> a safe ?redirectBack= (linked
 * pages) -> the caller's fallback (deep links).
 */
export type Origin = { pathname: string; search?: string; label?: string };
export type OriginState = { from: Origin };
export type BackTarget = { to: string; label: string | undefined; source: "state" | "redirectBack" | "fallback" };

export function isSafeInternalPath(value: unknown): value is string {
  if (typeof value !== "string" || value.length === 0 || value.length > 2048) return false;
  if (!value.startsWith("/")) return false;
  if (value.startsWith("//") || value.startsWith("/\\")) return false;
  return !/[\u0000-\u001f]/.test(value);
}

export function withOrigin<T extends Record<string, unknown>>(
  location: { pathname: string; search?: string },
  label: string,
  extra?: T,
): (T & OriginState) | OriginState {
  const from: Origin = { pathname: location.pathname, search: location.search ?? "", label };
  return extra ? { ...extra, from } : { from };
}

export function readOrigin(state: unknown): Origin | null {
  if (!state || typeof state !== "object") return null;
  const from = (state as { from?: unknown }).from;
  if (!from || typeof from !== "object") return null;
  const { pathname, search, label } = from as Record<string, unknown>;
  if (!isSafeInternalPath(pathname)) return null;
  const safeSearch = typeof search === "string" && (search === "" || search.startsWith("?")) ? search : "";
  return { pathname, search: safeSearch, label: typeof label === "string" && label.trim() ? label : undefined };
}

export function resolveBack(input: {
  state: unknown;
  search: string;
  currentPathname: string;
  fallback: { to: string; label: string };
}): BackTarget {
  const { state, search, currentPathname, fallback } = input;
  const origin = readOrigin(state);
  if (origin && origin.pathname !== currentPathname) {
    return { to: `${origin.pathname}${origin.search ?? ""}`, label: origin.label, source: "state" };
  }
  const redirectBack = new URLSearchParams(search).get("redirectBack");
  if (isSafeInternalPath(redirectBack) && redirectBack.split("?")[0] !== currentPathname) {
    return { to: redirectBack, label: undefined, source: "redirectBack" };
  }
  return { to: fallback.to, label: fallback.label, source: "fallback" };
}
```

```ts
// explorers-earth/src/lib/navigation/categoryRoutes.ts
/** Seven non-Places categories (Places joins the shared route in WP6). */
export type CategoryKey = "movies" | "books" | "games" | "apps" | "products" | "people" | "guides";

export type CategoryRoute = {
  key: CategoryKey;
  /** i18n key for the category label used by Back ("Back to Movies & Shows"). */
  labelKey: `dashboard.wp1.back.categories.${CategoryKey}`;
  listParam: ":listId" | ":guideId";
  /** Path segment(s) after the list id for the add page; null when not a simple suffix. */
  addSuffix: string | null;
};

export const CATEGORY_ROUTES: readonly CategoryRoute[] = Object.freeze([
  { key: "movies", labelKey: "dashboard.wp1.back.categories.movies", listParam: ":listId", addSuffix: "add" },
  { key: "books", labelKey: "dashboard.wp1.back.categories.books", listParam: ":listId", addSuffix: "add" },
  { key: "games", labelKey: "dashboard.wp1.back.categories.games", listParam: ":listId", addSuffix: "add" },
  { key: "apps", labelKey: "dashboard.wp1.back.categories.apps", listParam: ":listId", addSuffix: "add" },
  { key: "products", labelKey: "dashboard.wp1.back.categories.products", listParam: ":listId", addSuffix: "add" },
  { key: "people", labelKey: "dashboard.wp1.back.categories.people", listParam: ":listId", addSuffix: "add" },
  { key: "guides", labelKey: "dashboard.wp1.back.categories.guides", listParam: ":guideId", addSuffix: "sections/new" },
]);

const byKey = (key: CategoryKey): CategoryRoute => {
  const route = CATEGORY_ROUTES.find(c => c.key === key);
  if (!route) throw new Error(`Unknown category ${key}`);
  return route;
};

export const categoryPath = (key: CategoryKey): string => `/recommendations/${byKey(key).key}`;
export const listPath = (key: CategoryKey, listId: string): string => `${categoryPath(key)}/${encodeURIComponent(listId)}`;
export const addItemPath = (key: CategoryKey, listId: string): string =>
  `${listPath(key, listId)}/${byKey(key).addSuffix ?? "add"}`;
export const categoryLabelKey = (key: CategoryKey) => byKey(key).labelKey;
```

```ts
// explorers-earth/src/lib/navigation/useOriginBack.ts
import { useCallback, useMemo } from "react";
import { useLocation, useNavigate, type NavigateOptions } from "react-router-dom";
import { resolveBack, type BackTarget } from "./originBack";

export type BackFallback = { to: string; label: string };

export function useOriginBack(fallback: BackFallback): BackTarget & { goBack: (options?: NavigateOptions) => void } {
  const location = useLocation();
  const navigate = useNavigate();
  const target = useMemo(
    () => resolveBack({ state: location.state, search: location.search, currentPathname: location.pathname, fallback }),
    [location.state, location.search, location.pathname, fallback.to, fallback.label],
  );
  const goBack = useCallback((options?: NavigateOptions) => navigate(target.to, options), [navigate, target.to]);
  return { ...target, goBack };
}
```
