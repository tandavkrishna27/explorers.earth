# Wave 5: First-visit samples (WP11, SMP-01..10) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: use superpowers:subagent-driven-development (preferred) or superpowers:executing-plans to carry out this plan task by task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A new account finishes onboarding and finds 2 sample lists in each of the seven list categories and one sample Trip guide, picked by its onboarding country (India or Global). Samples are badged, browsable and reorderable. They can never be published. "Make it mine" turns one into a normal draft in place. A deleted sample never comes back.

**Architecture:** Versioned JSON fixtures live in `tunes/server/fixtures/samples/{india,global}/v1/`. They are imported statically, so esbuild bundles them into `dist/server`, which is the same pattern as `application/richNote.ts:5` with `quillEmojiData.json`. A `SampleSeeder` runs after the onboarding update commits. It works **per sample**, not per account: every list and the guide has a `sample_key` `'<category>/<list key>'`, and S3/0071's `sample_seeding_receipts` holds one **permanent** row per (account, sample_key) with `fixture_set` (`india|global`) and `fixture_version` (`v1`). A key that has a receipt is never seeded again, whether the list was kept, made mine (`claimed_at` set) or deleted (`collection_id` NULL), which makes deletes permanent. A key without a receipt is created through the existing repository commands (`resolve*Entity`, `createCollection`, `createRecommendation`, `writeCategoryTopPicks`; for the guide `writeGuideDetails` + `createGuideSection` with `guide-section-blocks/v2` items), then S3's `recordSampleSeeded` writes the receipt. Those commands use deterministic idempotency keys, so a crashed seed resumes by replaying and never duplicates. S3 ships the schema, `tunes/shared/explorersSampleContract.ts` and `tunes/server/repositories/sampleRepository.ts` (`readSampleReceipts`, `recordSampleSeeded`, `claimSample`, `isSampleCollection`); WP11 consumes them and recreates none of them. The recommendation repository is the only place that sets `collections.is_sample` and `recommendations.is_sample`; it refuses every publish path with `SAMPLE_PUBLISH_REFUSAL` (422). S3's CHECKs `collections_sample_never_public` and `recommendations_sample_never_published` back this up (23514, mapped to the same 422), and every public read also filters `is_sample`. Make it mine is S3's `claimSample` plus a revision bump. In the SPA, `SampleBadge` appears on list cards and Top Picks, the publish pill is disabled with "Samples can't be published", and the list page gets the Make it mine and Delete actions.

**Tech Stack:** Express + TypeScript, PostgreSQL 15 (pg), zod/v3 shared contracts, vitest (unit + `vitest.integration.config.ts`), React 18 + TanStack Query in explorers-earth, Playwright.

```ts
import { z } from "zod/v3";
import { resolveManualMovieSchema } from "../../shared/explorersMovieContract";
import { resolveManualBookSchema } from "../../shared/explorersBookContract";
import { resolveManualGameSchema } from "../../shared/explorersContract";

import { resolveManualPlaceSchema } from "../../shared/explorersPlaceContract";
import indiaPlaces from "../fixtures/samples/india/v1/places.json";
import indiaMovies from "../fixtures/samples/india/v1/movies.json";
import indiaBooks from "../fixtures/samples/india/v1/books.json";
import indiaGames from "../fixtures/samples/india/v1/games.json";
import indiaApps from "../fixtures/samples/india/v1/apps.json";
import indiaProducts from "../fixtures/samples/india/v1/products.json";
import indiaPeople from "../fixtures/samples/india/v1/people.json";
import indiaGuides from "../fixtures/samples/india/v1/guides.json";
import globalPlaces from "../fixtures/samples/global/v1/places.json";
import globalMovies from "../fixtures/samples/global/v1/movies.json";
import globalBooks from "../fixtures/samples/global/v1/books.json";
import globalGames from "../fixtures/samples/global/v1/games.json";
import globalApps from "../fixtures/samples/global/v1/apps.json";
import globalProducts from "../fixtures/samples/global/v1/products.json";
import globalPeople from "../fixtures/samples/global/v1/people.json";
import globalGuides from "../fixtures/samples/global/v1/guides.json";

import { SAMPLE_FIXTURE_SETS, sampleKeySchema, type SampleFixtureSet, type SampleKey } from "../../shared/explorersSampleContract";
import { guideDayItemSchema, guideTimeLabelSchema, type GuideDayItem } from "../../shared/explorersGuideContract";

/** S3 `fixture_version` ('v1'). Bump with a new `v2/` folder; never edit a shipped version (receipts record which one each key came from). */
export const SAMPLE_FIXTURE_VERSION = "v1" as const;
export type SampleSet = SampleFixtureSet; // 'india' | 'global', from S3
export const SAMPLE_LIST_CATEGORIES = ["places", "movies", "books", "games", "apps", "products", "people"] as const;
export type SampleListCategory = (typeof SAMPLE_LIST_CATEGORIES)[number];
/** S3 `sample_key` = '<category>/<slug>'; the fixture's list key is the slug. */
export const sampleKeyFor = (category: SampleListCategory | "guides", listKey: string): SampleKey => sampleKeySchema.parse(`${category}/${listKey}`);

const detailsSchemas = {
  places: resolveManualPlaceSchema.shape.details,
  movies: resolveManualMovieSchema.shape.details,
  books: resolveManualBookSchema.shape.details,
  games: resolveManualGameSchema.shape.details,
  apps: resolveManualAppSchema.shape.details,
  products: resolveManualProductSchema.shape.details,
  people: resolveManualPersonSchema.shape.details,
} as const;

const listKey = z.string().regex(/^[a-z0-9]+(-[a-z0-9]+)*$/).max(80);
const note = z.string().min(1).max(400);
const rating = z.number().int().min(1).max(10);
const googlePlaceId = z.string().regex(/^[A-Za-z0-9_-]{10,}$/).nullable();

const itemSchema = <C extends SampleListCategory>(category: C) => z.object({
  note, rating,
  details: detailsSchemas[category] as (typeof detailsSchemas)[C],
  ...(category === "places" ? { googlePlaceId } : {}),
}).strict();

const listSchema = <C extends SampleListCategory>(category: C) => z.object({
  key: listKey, title: z.string().min(1).max(200), description: z.string().min(1).max(500),
  topPicks: z.boolean(), items: z.array(itemSchema(category)).min(1).max(12),
}).strict();

export const sampleFixtureFileSchema = z.discriminatedUnion("category", SAMPLE_LIST_CATEGORIES.map((category) => z.object({
  set: z.enum(SAMPLE_FIXTURE_SETS), version: z.literal(SAMPLE_FIXTURE_VERSION), category: z.literal(category),
  lists: z.array(listSchema(category)).max(3),
}).strict()) as unknown as [z.ZodTypeAny, z.ZodTypeAny, ...z.ZodTypeAny[]]);

// Guide fixture = the S3/0069 v2 day model in a compact authoring form: each day becomes ONE
// guide_sections row (display_order = day index, block_version 'guide-section-blocks/v2') whose
// blocks.items is this ordered list. `toGuideDayItem` expands a compact place into S3's full
// GuideDayPlace; the fixture test proves every expanded item parses guideDayItemSchema.
// 'listItem' is not used in v1 fixtures (its ref needs a real recommendation id).
const localId = z.string().regex(/^[a-z0-9]+(-[a-z0-9]+)*$/).max(100);
const timeLabel = guideTimeLabelSchema.nullable();
const guidePlaceItemSchema = z.object({
  kind: z.literal("place"), localId, timeLabel, tips: note, googlePlaceId, details: detailsSchemas.places,
}).strict();
const guideNoteItemSchema = z.object({ kind: z.literal("note"), localId, timeLabel, text: note }).strict();
const guideFixtureItemSchema = z.discriminatedUnion("kind", [guidePlaceItemSchema, guideNoteItemSchema]);
export type SampleGuideFixtureItem = z.infer<typeof guideFixtureItemSchema>;

export const sampleGuideFileSchema = z.object({
  set: z.enum(SAMPLE_FIXTURE_SETS), version: z.literal(SAMPLE_FIXTURE_VERSION), category: z.literal("guides"),
  guide: z.object({
    key: listKey, title: z.string().min(1).max(200), description: z.string().min(1).max(500),
    guideType: z.literal("itinerary"), numberOfDays: z.number().int().min(1).max(14),
    days: z.array(z.object({ day: z.number().int().min(1), items: z.array(guideFixtureItemSchema).min(1).max(8) }).strict()).min(1).max(14),
  }).strict(),
}).strict();

/** Compact fixture item -> S3 v2 GuideDayItem. Field list must equal S3's guideDayPlaceSchema (re-verify list). */
export function toGuideDayItem(item: SampleGuideFixtureItem): GuideDayItem {
  if (item.kind === "note") return guideDayItemSchema.parse({ kind: "note", localId: item.localId, timeLabel: item.timeLabel, text: item.text });
  const d = item.details;
  return guideDayItemSchema.parse({ kind: "place", localId: item.localId, timeLabel: item.timeLabel, place: {
    localId: item.localId, placeId: item.googlePlaceId, name: d.title, formattedAddress: d.formattedAddress ?? null,
    geometry: d.latitude != null && d.longitude != null ? { location: { lat: d.latitude, lng: d.longitude } } : null,
    types: [], tips: item.tips, photos: [], priceLevel: null, priceRange: null, customBudget: null,
    budgetAmount: null, budgetCurrency: null, source: item.googlePlaceId ? "google" : "manual", verified: item.googlePlaceId !== null,
  } });
}

export type SampleListFile<C extends SampleListCategory = SampleListCategory> = {
  set: SampleSet; version: typeof SAMPLE_FIXTURE_VERSION; category: C; lists: Array<z.infer<ReturnType<typeof listSchema<C>>>>;
};
export type SampleGuideFile = z.infer<typeof sampleGuideFileSchema>;
export type LoadedSampleSet = { set: SampleSet; lists: { [C in SampleListCategory]: SampleListFile<C> }; guide: SampleGuideFile };

const raw: Record<SampleSet, Record<SampleListCategory | "guides", unknown>> = {
  india: { places: indiaPlaces, movies: indiaMovies, books: indiaBooks, games: indiaGames, apps: indiaApps, products: indiaProducts, people: indiaPeople, guides: indiaGuides },
  global: { places: globalPlaces, movies: globalMovies, books: globalBooks, games: globalGames, apps: globalApps, products: globalProducts, people: globalPeople, guides: globalGuides },
};

const cache = new Map<SampleSet, LoadedSampleSet>();
/** Parses once per process; a malformed fixture throws at first use and is caught by sampleFixtures.test.ts in CI. */
export function loadSampleSet(set: SampleSet): LoadedSampleSet {
  const hit = cache.get(set);
  if (hit) return hit;
  const lists = Object.fromEntries(SAMPLE_LIST_CATEGORIES.map((category) => {
    const file = sampleFixtureFileSchema.parse(raw[set][category]) as SampleListFile;
    if (file.set !== set || file.category !== category) throw new Error(`Sample fixture ${set}/${category} is mislabelled`);
    return [category, file];
  })) as LoadedSampleSet["lists"];
  const guide = sampleGuideFileSchema.parse(raw[set].guides);
  if (guide.set !== set) throw new Error(`Sample fixture ${set}/guides is mislabelled`);
  const loaded = { set, lists, guide };
  cache.set(set, loaded);
  return loaded;
}
```

```ts
// tunes/scripts/resolve-sample-place-ids.ts
import { readFileSync, writeFileSync } from "node:fs";
import { resolve } from "node:path";

type PlaceDetails = { title: string; formattedAddress?: string | null; latitude?: number | null; longitude?: number | null };
type PlaceItem = { googlePlaceId: string | null; details: PlaceDetails };
// Guide days mix v2 'place' and 'note' items; only place items carry details.
type GuideItem = ({ kind: "place" } & PlaceItem) | { kind: "note" };
type Fixture = { lists?: Array<{ items: PlaceItem[] }>; guide?: { days: Array<{ items: GuideItem[] }> } };
type Options = { key: string; fetcher?: typeof fetch };

const endpoint = "https://places.googleapis.com/v1/places:searchText";

/** Pure over its fetcher: resolves only items whose id is still null, so a re-run costs nothing. */
export async function resolvePlacesInFixture<F extends Fixture>(fixture: F, { key, fetcher = fetch }: Options) {
  const unresolved: string[] = [];
  const items = [...(fixture.lists ?? []).flatMap((l) => l.items), ...(fixture.guide?.days ?? []).flatMap((d) => d.items.flatMap((i) => (i.kind === "place" ? [i] : [])))];
  for (const item of items) {
    if (item.googlePlaceId) continue;
    const textQuery = [item.details.title, item.details.formattedAddress].filter(Boolean).join(", ");
    const response = await fetcher(endpoint, { method: "POST", body: JSON.stringify({ textQuery }), headers: {
      "Content-Type": "application/json", "X-Goog-Api-Key": key, "X-Goog-FieldMask": "places.id,places.formattedAddress,places.location" } });
    const hit = ((await response.json()) as { places?: Array<{ id: string; formattedAddress?: string; location?: { latitude: number; longitude: number } }> }).places?.[0];
    if (!hit?.id || !hit.location) { unresolved.push(item.details.title); continue; }
    item.googlePlaceId = hit.id;
    item.details.formattedAddress = hit.formattedAddress ?? item.details.formattedAddress ?? null;
    item.details.latitude = hit.location.latitude;
    item.details.longitude = hit.location.longitude;
  }
  return { fixture, unresolved };
}

if (process.argv[1]?.replace(/\\/g, "/").endsWith("/scripts/resolve-sample-place-ids.ts")) {
  const key = process.env.GOOGLE_PLACES_AUTHORING_KEY;
  if (!key) { console.error("Set GOOGLE_PLACES_AUTHORING_KEY in your own shell; it is never committed."); process.exit(2); }
  const files = process.argv.slice(2).filter((a) => a.endsWith(".json"));
  let failed = false;
  for (const file of files) {
    const path = resolve(file);
    const { fixture, unresolved } = await resolvePlacesInFixture(JSON.parse(readFileSync(path, "utf8")), { key });
    writeFileSync(path, `${JSON.stringify(fixture, null, 2)}\n`);
    if (unresolved.length) { failed = true; console.error(`${file}: unresolved ${unresolved.join("; ")}`); }
  }
  process.exit(failed ? 1 : 0);
}
```

```ts
// Onboarding country is free text; S3's sampleCountryToFixtureSet only knows the ISO code 'IN'.
// WP11 normalises India's names to 'IN' and leaves everything else as typed (=> global).
const INDIA = new Set(["india", "in", "ind", "bharat", "भारत", "republic of india"]);
export function normalizeSampleCountry(country: string | null | undefined): string | null {
  const value = (country ?? "").trim();
  if (!value) return null;
  return INDIA.has(value.toLowerCase()) ? "IN" : value;
}
export function fixtureSetForCountry(country: string | null | undefined): SampleFixtureSet {
  return sampleCountryToFixtureSet(normalizeSampleCountry(country));

type Address = { country?: string } | null | undefined;
export function countryFromAccount(account: { additionalAddresses?: Address[] | null; primaryAddress?: Address }): string | null {
  for (const address of [...(account.additionalAddresses ?? []), account.primaryAddress]) {
    const country = address?.country?.trim();
    if (country) return country;
  }
  return null;

const escapeHtml = (value: string) => value.replace(/[&<>"']/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" })[c]!);
const slugify = (title: string) => title.toLowerCase().normalize("NFKD").replace(/[^\p{Letter}\p{Number}]+/gu, "-")
  .replace(/[^a-z0-9-]/g, "").replace(/^-+|-+$/g, "").slice(0, 80) || "sample";
const topPickCategories = new Set<SampleListCategory>(["movies", "books", "games", "apps", "products", "people"]);

/** Every sample key a fixture set produces (lists + the guide). */
export function sampleKeysForSet(set: SampleFixtureSet): SampleKey[] {
  const sample = loadSampleSet(set);
  return [...SAMPLE_LIST_CATEGORIES.flatMap((c) => sample.lists[c].lists.map((l) => sampleKeyFor(c, l.key))), sampleKeyFor("guides", sample.guide.guide.key)];
}

/**
 * SMP-08 on S3/0071. Seeding is per sample key ('<category>/<slug>'). A key with a row in
 * sample_seeding_receipts is NEVER seeded again (kept, made mine, or deleted - SMP-05). A key
 * without one is created through repository commands with deterministic idempotency keys, then
 * S3's recordSampleSeeded writes the permanent receipt. A crash between the two replays the same
 * commands (same ids, nothing duplicated) and records the receipt on the next run.
 */
export class SampleSeeder {
  constructor(private readonly db: Pool, private readonly repository = new ExplorersRecommendationRepository(db)) {}

  async seedForAccount(accountId: string, country: string | null): Promise<"seeded" | "already-seeded"> {
    const receipts = await readSampleReceipts(this.db, accountId);
    // A resumed seed keeps the fixture set it first chose, even if the country changed since.
    const fixtureSet: SampleFixtureSet = receipts[0]?.fixtureSet ?? fixtureSetForCountry(country);
    const done = new Set(receipts.map((r) => r.sampleKey));
    if (sampleKeysForSet(fixtureSet).every((k) => done.has(k))) return "already-seeded";
    const sample = loadSampleSet(fixtureSet);
    const key = (...parts: Array<string | number>) => ["sample", SAMPLE_FIXTURE_VERSION, ...parts].join("-").replace(/\//g, "-");
    let seededAny = false;

    for (const category of SAMPLE_LIST_CATEGORIES) {
      const pins: Array<{ recommendationId: string; collectionId: string }> = [];
      for (const list of sample.lists[category].lists) {
        const sampleKey = sampleKeyFor(category, list.key);
        if (done.has(sampleKey)) continue;
        const created = await this.repository.createCollection(accountId, { category, title: list.title, slug: slugify(list.title),
          description: list.description, visibility: "private", publicationState: "draft", isSample: true }, key(sampleKey));
        let revision = created.revision;
        for (const [index, item] of list.items.entries()) {
          const entityId = await this.resolve(accountId, category, item as never, key(sampleKey, index, "entity"));
          const recommendation = await this.repository.createRecommendation(accountId, { category, entityId, collectionId: created.id,
            expectedCollectionRevision: revision, userRating: item.rating, publicationState: "draft", isSample: true,
            note: richNoteFromEditor(`<p>${escapeHtml(item.note)}</p>`) }, key(sampleKey, index));
          revision = await this.collectionRevision(created.id);
          if (list.topPicks && pins.length < 5) pins.push({ recommendationId: recommendation.id, collectionId: created.id });
        }
        seededAny = (await this.recordReceipt(accountId, sampleKey, category, fixtureSet, created.id)) || seededAny;
      }
      if (pins.length && topPickCategories.has(category)) await this.pin(accountId, category as never, pins, key(category, "top-picks"));
    }
    const guideKey = sampleKeyFor("guides", sample.guide.guide.key);
    if (!done.has(guideKey)) {
      const guideId = await this.seedGuide(accountId, sample.guide.guide, key(guideKey));
      seededAny = (await this.recordReceipt(accountId, guideKey, "guides", fixtureSet, guideId)) || seededAny;
    }
    return seededAny ? "seeded" : "already-seeded";
  }

  /**
   * S3's recordSampleSeeded: false = a concurrent seeder recorded this key first. Deterministic
   * idempotency keys mean both seeders created the SAME list, so normally there is nothing to undo;
   * if the recorded receipt points at a different list, archive ours (S3: "caller must roll back").
   */
  private async recordReceipt(accountId: string, sampleKey: SampleKey, category: SampleListCategory | "guides", fixtureSet: SampleFixtureSet, collectionId: string): Promise<boolean> {
    const client = await this.db.connect();
    try {
      await client.query("BEGIN");
      const recorded = await recordSampleSeeded(client, { accountId, sampleKey, category, fixtureSet, fixtureVersion: SAMPLE_FIXTURE_VERSION, collectionId });
      await client.query("COMMIT");
      if (!recorded) {
        const winner = (await readSampleReceipts(this.db, accountId)).find((r) => r.sampleKey === sampleKey);
        if (winner && winner.collectionId !== collectionId) {
          const revision = await this.collectionRevision(collectionId);
          await this.repository.archiveCollection(accountId, collectionId, revision, `sample-${SAMPLE_FIXTURE_VERSION}-${sampleKey.replace("/", "-")}-rollback`);
        }
      }
      return recorded;
    } catch (error) { await client.query("ROLLBACK"); throw error; } finally { client.release(); }
  }

  private async collectionRevision(id: string): Promise<number> {
    return Number((await this.db.query("SELECT revision::text AS revision FROM collections WHERE id=$1", [id])).rows[0].revision);
  }

  private async resolve(accountId: string, category: SampleListCategory, item: { details: Record<string, unknown> & { title: string }; googlePlaceId?: string | null }, entityKey: string): Promise<string> {
    switch (category) {
      case "places": return this.resolvePlace(accountId, item, entityKey);
      case "movies": return (await this.repository.resolveMovieEntity(accountId, { kind: "manual", category, details: item.details }, entityKey, async () => null as never)).id;
      case "books": return (await this.repository.resolveBookEntity(accountId, { kind: "manual", category, details: item.details as never }, entityKey)).id;
      case "games": return (await this.repository.resolveGameEntity(accountId, { details: item.details as never }, entityKey)).id;
      case "apps": return (await this.repository.resolveAppEntity(accountId, { kind: "manual", category, details: item.details }, entityKey)).id;
      case "products": return (await this.repository.resolveProductEntity(accountId, { kind: "manual", category, details: item.details }, entityKey)).id;
      case "people": return (await this.repository.resolvePersonEntity(accountId, { kind: "manual", category, details: item.details }, entityKey)).id;
    }
  }

  /** A Google-resolved place is one shared entity for every account (SMP-08: stored ids, no runtime cost). */
  private async resolvePlace(accountId: string, item: { details: Record<string, unknown> & { title: string }; googlePlaceId?: string | null }, entityKey: string): Promise<string> {
    if (!item.googlePlaceId) return (await this.repository.resolvePlaceEntity(accountId, { kind: "manual", category: "places", details: item.details }, entityKey)).id;
    const entity = await this.repository.resolveProviderEntity({ kind: "place", title: item.details.title, provider: "google_places", externalKind: "place", externalId: item.googlePlaceId });
    await this.db.query(`INSERT INTO place_entity_details(entity_id,formatted_address,latitude,longitude)
      VALUES($1,$2,$3,$4) ON CONFLICT(entity_id) DO NOTHING`,
      [entity.id, item.details.formattedAddress ?? null, item.details.latitude ?? null, item.details.longitude ?? null]);
    return entity.id;
  }

  private async pin(accountId: string, category: "books" | "movies" | "games" | "apps" | "products" | "people", orderedPins: Array<{ recommendationId: string; collectionId: string }>, pinKey: string) {
    const state = await this.db.query<{ category_revision: string; pin_revision: string | null }>(`SELECT
      coalesce((SELECT revision::text FROM account_category_content_state WHERE account_id=$1 AND category=$2),'0') AS category_revision,
      (SELECT revision::text FROM account_category_pin_state WHERE account_id=$1 AND category=$2) AS pin_revision`, [accountId, category]);
    const row = state.rows[0]!;
    await this.repository.writeCategoryTopPicks(accountId, category, { expectedCategoryRevision: row.category_revision,
      expectedPinRevision: row.pin_revision === null ? null : Number(row.pin_revision), orderedPins }, pinKey, true);
  }

  /**
   * SMP-10 on S3/0069: one guide_sections row per day (display_order = day index,
   * block_version 'guide-section-blocks/v2'), blocks.items = the day's ordered v2 items.
   * Idempotent on replay: sections are written only while the guide has none.
   */
  private async seedGuide(accountId: string, guide: ReturnType<typeof loadSampleSet>["guide"]["guide"], collectionKey: string): Promise<string> {
    const created = await this.repository.createCollection(accountId, { category: "guides", title: guide.title, slug: slugify(guide.title),
      description: guide.description, visibility: "private", publicationState: "draft", isSample: true }, collectionKey);
    const client = await this.db.connect();
    try {
      await client.query("BEGIN");
      const existing = await client.query("SELECT 1 FROM guide_sections WHERE collection_id=$1 LIMIT 1", [created.id]);
      if (!existing.rows[0]) {
        await writeGuideDetails(client, created.id, accountId, { guideType: guide.guideType, multiCity: false, numberOfDays: guide.numberOfDays,
          estimatedBudget: null, budgetCurrency: null, budgetType: null, budgetTier: null, startDate: null, bestTimeToVisit: [], categories: [],
          tags: [], tipsNotes: null, place: {}, locationEntityId: null });
        for (const day of [...guide.days].sort((a, b) => a.day - b.day)) {
          const blocks = guideSectionBlocksSchema.parse({ ...guideEmptySectionBlocks, items: day.items.map(toGuideDayItem), budget: { entries: [] } });
          await createGuideSection(client, created.id, accountId, await this.categoryRevision(client, accountId, "guides"), { title: "", description: null, blocks });
        }
        const versions = await client.query("SELECT DISTINCT block_version FROM guide_sections WHERE collection_id=$1", [created.id]);
        if (versions.rows.some((r) => r.block_version !== GUIDE_SECTION_BLOCK_VERSION)) throw new Error("sample guide written with a non-v2 block version");
      }
      await client.query("COMMIT");
      return created.id;
    } catch (error) { await client.query("ROLLBACK"); throw error; } finally { client.release(); }
  }

  private async categoryRevision(db: Pick<PoolClient, "query">, accountId: string, category: string): Promise<number> {
    return Number((await db.query("SELECT coalesce((SELECT revision::text FROM account_category_content_state WHERE account_id=$1 AND category=$2),'0') AS r", [accountId, category])).rows[0].r);
  }
}
```

```ts
import { expect, test, type Page } from "@playwright/test";
import { signInFreshTestAccount } from "./setup/local-stack-auth";

const baseURL = process.env.FIRST_VISIT_BASE_URL;
test.skip(!baseURL, "FIRST_VISIT_BASE_URL is not set; this spec needs the local proof stack");

async function onboard(page: Page, country: string) {
  await signInFreshTestAccount(page);
  await page.goto("/onboarding");
  await page.getByLabel(/username/i).fill(`fv${Date.now()}`);
  await page.getByLabel(/account name/i).fill("First Visit");
  await page.getByLabel(/country/i).fill(country);
  await page.getByLabel(/city/i).fill(country === "India" ? "Mumbai" : "Paris");
  await page.getByLabel(/postal code/i).fill(country === "India" ? "400001" : "75001");
  await page.getByRole("button", { name: /finish|complete|continue/i }).last().click();
  await expect(page).not.toHaveURL(/onboarding/);
}

for (const [country, list, guide] of [["India", "Delhi heritage walk", "3 days in Goa"], ["France", "Paris landmarks", "48 hours in Paris"]] as const) {
```
