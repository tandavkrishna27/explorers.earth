```ts
// Stored ratings are integers 1-10 (tunes/shared/explorersContract.ts:59).
// The UI shows 5 stars in half steps, so one stored point is half a star.
export type StoredRating = number | null;

export function starsFromValue(value: StoredRating): number {
  if (value === null) return 0;
  if (!Number.isInteger(value) || value < 1 || value > 10) throw new RangeError(`rating ${value}`);
  return value / 2;
}

export function valueFromStars(stars: number): StoredRating {
  if (stars === 0) return null;
  const value = stars * 2;
  if (!Number.isInteger(value) || value < 1 || value > 10) throw new RangeError(`stars ${stars}`);
  return value;
}

export function stepValue(value: StoredRating, direction: 1 | -1): StoredRating {
  const next = (value ?? 0) + direction;
  if (next <= 0) return null;
  return Math.min(10, next);
}
```

```tsx
import { Star } from "lucide-react";
import type { KeyboardEvent } from "react";
import { starsFromValue, stepValue, type StoredRating } from "./starRatingScale";

type Props = { value: StoredRating; onChange: (value: StoredRating) => void; label: string; disabled?: boolean };

// One widget for every category (scope ADD-05 / DS-01). Each star has two hit areas (half steps).
export default function StarRating({ value, onChange, label, disabled = false }: Props) {
  const stars = starsFromValue(value);
  const pick = (stored: number) => { if (!disabled) onChange(stored === value ? null : stored); };
  const onKeyDown = (event: KeyboardEvent<HTMLDivElement>) => {
    if (disabled) return;
    const keys: Record<string, StoredRating> = {
      ArrowRight: stepValue(value, 1), ArrowUp: stepValue(value, 1),
      ArrowLeft: stepValue(value, -1), ArrowDown: stepValue(value, -1),
      Home: null, End: 10,
    };
    if (!(event.key in keys)) return;
    event.preventDefault();
    onChange(keys[event.key]);
  };
  return (
    <div
      role="slider" tabIndex={disabled ? -1 : 0} aria-label={label}
      aria-valuemin={0} aria-valuemax={10} aria-valuenow={value ?? 0}
      aria-valuetext={`${stars} of 5 stars`} aria-disabled={disabled || undefined}
      onKeyDown={onKeyDown} className="inline-flex gap-1 rounded-md focus-visible:outline focus-visible:outline-2 focus-visible:outline-blue-400"
    >
      {[1, 2, 3, 4, 5].map((n) => {
        const fill = stars >= n ? "full" : stars >= n - 0.5 ? "half" : "empty";
        return (
          <span key={n} className="relative h-11 w-11" aria-hidden={false}>
            <Star aria-hidden className="absolute inset-0 m-auto h-7 w-7 text-gray-600" />
            {fill !== "empty" && (
              <span className={`absolute inset-y-0 left-0 overflow-hidden ${fill === "half" ? "w-1/2" : "w-full"}`}>
                <Star aria-hidden className="absolute left-0 top-0 bottom-0 m-auto ml-2 h-7 w-7 fill-amber-500 text-amber-500" />
              </span>
            )}
            <button type="button" tabIndex={-1} aria-label={`${n - 0.5} stars`} disabled={disabled}
              onClick={() => pick(n * 2 - 1)} className="absolute inset-y-0 left-0 w-1/2" />
            <button type="button" tabIndex={-1} aria-label={`${n} stars`} disabled={disabled}
              onClick={() => pick(n * 2)} className="absolute inset-y-0 right-0 w-1/2" />
          </span>
        );
      })}
    </div>
  );
}
```

Notes: each star cell is 44px (`h-11 w-11`) for the touch-target rule; the hit buttons are `tabIndex={-1}` so keyboard users get one tab stop (the slider). Disabled buttons do not fire `onClick` in jsdom, which the last test relies on.

```ts
import { useCallback, useState } from "react";

export type AddItemMode = "search" | "manual" | "picked";

export function useAddItemMode<T>(options: { initialPicked?: T | null } = {}) {
  const [mode, setMode] = useState<AddItemMode>(options.initialPicked ? "picked" : "search");
  const [query, setQuery] = useState("");
  const [manualName, setManualName] = useState("");
  const [picked, setPicked] = useState<T | null>(options.initialPicked ?? null);

  const goManual = useCallback(() => { setManualName((name) => name || query); setMode("manual"); }, [query]);
  const backToSearch = useCallback(() => setMode("search"), []);
  const pick = useCallback((item: T) => { setPicked(item); setMode("picked"); }, []);
  const change = useCallback(() => { setPicked(null); setMode("search"); }, []);
  const reset = useCallback(() => { setMode("search"); setQuery(""); setManualName(""); setPicked(null); }, []);

  return { mode, query, setQuery, manualName, setManualName, picked, goManual, backToSearch, pick, change, reset };
}
```

```ts
export type TaxonomyTerm = { id: string; parentId: string | null; position: number; label: string };

const fold = (text: string) => text.normalize("NFD").replace(/\p{Diacritic}/gu, "").toLowerCase();
const byPosition = (a: TaxonomyTerm, b: TaxonomyTerm) => a.position - b.position;

export const rootsOf = (terms: TaxonomyTerm[]) => terms.filter((t) => t.parentId === null).sort(byPosition);
export const childrenOf = (terms: TaxonomyTerm[], rootId: string) => terms.filter((t) => t.parentId === rootId).sort(byPosition);

export function filterTerms(terms: TaxonomyTerm[], query: string): TaxonomyTerm[] {
  const q = fold(query.trim());
  if (!q) return terms;
  const starts: TaxonomyTerm[] = [];
  const contains: TaxonomyTerm[] = [];
  for (const term of terms) {
    const label = fold(term.label);
    if (label.startsWith(q) || label.split(/[\s&/-]+/).some((w) => w.startsWith(q))) starts.push(term);
    else if (label.includes(q)) contains.push(term);
  }
  return [...starts, ...contains];
}
```

```tsx
export type TaxonomyValue = { categoryId: string | null; subcategoryId: string | null };

function Field({ label, options, selectedId, storedLabel, disabled, onPick }: {
  label: string; options: TaxonomyTerm[]; selectedId: string | null; storedLabel?: string; disabled?: boolean; onPick: (id: string | null) => void;
}) {

  const selected = options.find((o) => o.id === selectedId);
```

```ts
export type LatLng = { lat: number; lng: number };

const DECIMAL = String.raw`(-?\d{1,3}(?:\.\d+)?)\s*°?\s*([NSEW])?`;
const PAIR = new RegExp(String.raw`^\s*${DECIMAL}\s*[,\s]\s*${DECIMAL}\s*$`, "i");
const MAPS_AT = /@(-?\d{1,3}\.\d+),(-?\d{1,3}\.\d+)/;

const signed = (value: string, hemisphere: string | undefined) =>
  (hemisphere && /[SW]/i.test(hemisphere) ? -1 : 1) * Math.abs(Number(value));

export function parseCoordinates(input: string): LatLng | null {
  const text = input.trim();
  const at = MAPS_AT.exec(text);
  const match = at ? [text, at[1], undefined, at[2], undefined] : PAIR.exec(text);
  if (!match) return null;
  const lat = signed(match[1]!, match[2] ?? undefined);
  const lng = signed(match[3]!, match[4] ?? undefined);
  if (!Number.isFinite(lat) || !Number.isFinite(lng)) return null;
  if (Math.abs(lat) > 90 || Math.abs(lng) > 180) return null;
  if (lat === 0 && lng === 0) return null;
  return { lat, lng };
}
```

```ts
import type { ItemSearchResult } from "./ItemSearch";
import type { LatLng } from "./parseCoordinates";

export type NearbyLookup = (cell: LatLng) => Promise<ItemSearchResult[]>;

const round = (value: number) => Math.round(value * 1000) / 1000;

export function createNearbyCache(lookup: NearbyLookup, limit = 50) {
  const cells = new Map<string, Promise<ItemSearchResult[]>>();
  return {
    get(point: LatLng): Promise<ItemSearchResult[]> {
      const cell = { lat: round(point.lat), lng: round(point.lng) };
      const key = `${cell.lat},${cell.lng}`;
      const hit = cells.get(key);
      if (hit) { cells.delete(key); cells.set(key, hit); return hit; }
      const pending = lookup(cell).catch((error) => { cells.delete(key); throw error; });
      cells.set(key, pending);
      if (cells.size > limit) cells.delete(cells.keys().next().value!);
      return pending;
    },
  };
}
```
