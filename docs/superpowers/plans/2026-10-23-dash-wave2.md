# Wave 2: Schema S2 + Places, Movies/Books/Games, Apps/Products/People Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship schema PR S2 (migrations 0067 curated taxonomy catalogs, 0068 product/app/game/people data-shape changes) and then three parallel category packages that move every category's add/edit flow onto the Wave 1 templates: WP6 Places, WP7 Movies/Books/Games with curated term lists, WP8 Apps/Products/People with a shared paste-a-link prefill service and iTunes keyword search.

**Architecture:** S2 extends the existing Places/People taxonomy machinery (`taxonomy_catalogs`, reviewed import + receipts, `recommendation_taxonomy`) to five more categories as FLAT catalogs (one term, position 0); Movies keeps its 0037 TMDB-mapped vocabulary and is configured in place. 0068 is additive (new nullable/defaulted columns + one backfill); no column is dropped. Feature packages consume only merged contracts. The link preview is a new synchronous, owner-initiated, SSRF-guarded server service (it does not reuse the async App enrichment worker, whose fetcher requires a host allowlist).

**Tech Stack:** PostgreSQL 15 raw SQL migrations, Express + zod/v3 shared contracts (`tunes/shared`), parse5 (already a dependency of `appEnrichmentProviderCore.mjs`), React 18 + Vite SPA, Vitest (contained runner), Playwright.

Paths: SPA = `explorers-earth/src/`, API = `tunes/server/`, contracts = `tunes/shared/`, migrations = `tunes/migrations/`. Code truth = `origin/main` 8063651c plus the Wave 0/1 PRs listed below.

### Phase-plan corrections (verified on origin/main 8063651c)

1. **No server "places provider route" exists.** Place search/details run in the browser (`explorers-earth/src/lib/browserPlacesProvider.ts:24-34`, `features/Favorites/components/TopPlacesByCategory.tsx:344` already calls `places:searchNearby` from the browser). WP6 adds `nearby()` to `browserPlacesProvider` with a basic field mask, not a server endpoint.
2. **Movies cannot go through the reviewed importer.** 0037 seeded movies terms with random UUIDs (`0037:51`); the importer refuses unreserved current identities (`taxonomyCatalogImport.ts` plan: "Unreserved current identity deletion" / "Stable slug reassigned"). 0067 configures the movies catalog in place over the existing 27 TMDB-mapped terms; `movie_provider_genre_terms` already maps TMDB genres onto them, so `movieTerms` are already curated. No apply run for movies.
3. **Games `developer` needs a column.** `game_entity_details` (0064) has no developer column and no number was reserved; S2 adds it in 0068. IGDB search hits cannot carry it today because `SEARCH_FIELDS` (`gameCatalogContracts.ts:7`) omits `involved_companies`; WP7 adds `involved_companies.company.name,involved_companies.developer` there.
4. **DS-14 affiliate toggle and Apps "other platform links" need storage.** Both go into 0068 (`product_recommendation_context.affiliate_disclosed`, `app_entity_details.other_links`).
5. **Products cannot drop `productUrl`.** It is the shared entity identity (`0044:28` NOT NULL; excluded from display overrides in `explorersProductContract.ts`). "One link" = UI writes the same URL to `productUrl` (new entities) and `offer.buyUrl`; 0068 backfills `buy_url` from `product_url` where null; reads use `buyUrl ?? productUrl`.
6. **Specs cap is contract-only.** Existing rows may hold up to 200 specs (0061 trigger); a DB cap would make them unwritable. The write schema caps at 8; reads keep 200.
7. **Generated decoder.** Changing `createRecommendationSchema` (adding `curatedTermId`) requires regenerating `tunes/server/services/appEnrichmentDecoderFields.generated.mjs`, which the phase plan does not mention.

export type LatLng={lat:number;lng:number};
const part=String.raw`(-?\d{1,3}(?:\.\d+)?)\s*°?\s*([NSEW])?`;
const pattern=new RegExp(String.raw`^\s*${part}\s*[,\s]\s*${part}\s*$`,'i');

/** "15.49, 73.82", "15.49 73.82" or "15.49° N, 73.82° E" -> a point; anything else -> null. */
export function parseCoordinates(input:string):LatLng|null{
 const match=pattern.exec(input);if(!match)return null;
 const signed=(value:string,hemisphere:string|undefined,negative:string)=>{const n=Number(value);return hemisphere&&hemisphere.toUpperCase()===negative?-Math.abs(n):n;};
 const lat=signed(match[1],match[2],'S'),lng=signed(match[3],match[4],'W');
 if(match[2]&&!/[NS]/i.test(match[2])||match[4]&&!/[EW]/i.test(match[4]))return null;
 if(!Number.isFinite(lat)||!Number.isFinite(lng)||Math.abs(lat)>90||Math.abs(lng)>180)return null;
 return {lat,lng};
}

import {roundedAreaKey,type LatLng} from './coordinates';
export type NearbyPlace={id:string;label:string;lat:number;lng:number;address:string|null};
export type NearbyResult={source:'google'|'osm'|'none';places:NearbyPlace[]};
type Fetcher=(point:LatLng,signal?:AbortSignal)=>Promise<NearbyPlace[]>;
const storageKey=(key:string)=>`explorers.nearby.v1:${key}`;
const read=(key:string):NearbyResult|null=>{try{const raw=sessionStorage.getItem(storageKey(key));return raw?JSON.parse(raw) as NearbyResult:null;}catch{return null;}};
const write=(key:string,value:NearbyResult)=>{try{sessionStorage.setItem(storageKey(key),JSON.stringify(value));}catch{/* private mode: memory cache only */}};

export function createNearbyLookup({google,osm}:{google:Fetcher;osm:Fetcher}){
 const memory=new Map<string,Promise<NearbyResult>>();
 return (point:LatLng,signal?:AbortSignal):Promise<NearbyResult>=>{
  const key=roundedAreaKey(point);
  const stored=read(key);if(stored)return Promise.resolve(stored);
  const pending=memory.get(key);if(pending)return pending;
  const run=(async():Promise<NearbyResult>=>{
   let result:NearbyResult;
   try{result={source:'google',places:await google(point,signal)};}
   catch{try{result={source:'osm',places:await osm(point,signal)};}catch{result={source:'none',places:[]};}}
   if(result.source!=='none')write(key,result);
   return result;
  })();
  memory.set(key,run);run.catch(()=>memory.delete(key));
  return run;
 };
}

import {useEffect,useState} from 'react';
import {explorersApiClient} from './explorersApiClient';
import type {FlatCuratedCategory} from '../../../tunes/shared/explorersTaxonomyContract';

export type CuratedTerm={id:string;label:string;parentId:string|null;slug:string};
type State={terms:CuratedTerm[];unconfigured:boolean;loading:boolean;error:Error|null};
const cache=new Map<string,Promise<{terms:CuratedTerm[];unconfigured:boolean}>>();

export function useCuratedCatalog(category:FlatCuratedCategory|'movies'):State{
 const [state,setState]=useState<State>({terms:[],unconfigured:false,loading:true,error:null});
 useEffect(()=>{let live=true;
  const pending=cache.get(category)??explorersApiClient.getCompleteTaxonomyCatalog({category}).then(page=>({unconfigured:page.state!=='CONFIGURED',terms:page.items.map(i=>({id:i.id,label:i.label,parentId:i.parentId,slug:i.slug}))}));
  cache.set(category,pending);pending.catch(()=>cache.delete(category));
  pending.then(v=>{if(live)setState({...v,loading:false,error:null});},(error:Error)=>{if(live)setState(s=>({...s,loading:false,error}));});
  return ()=>{live=false;};},[category]);
 return state;
}

const words=(value:string)=>value.toLowerCase().split(/[^a-z0-9]+/).filter(Boolean);
/** First curated term whose label words all appear in one provider label (pre-selection only; the owner can change it). */
export function suggestTermFromLabels(terms:readonly CuratedTerm[],labels:readonly string[]):string|null{
 for(const label of labels){const have=new Set(words(label));const hit=terms.find(t=>words(t.label).filter(w=>w!=='and').every(w=>have.has(w)));if(hit)return hit.id;}
 return null;
}

export type Address={address:string;family:number};
export type Resolver=(host:string)=>Promise<Address[]>;
export class LinkPreviewBlocked extends Error{constructor(){super('LINK_PREVIEW_BLOCKED');}}
const defaultResolver:Resolver=host=>lookup(host,{all:true,verbatim:true});
/** One owner-pasted URL -> a pinned public address. Literal IPs and internal names are refused outright. */
export async function guardLinkPreviewTarget(raw:string,resolve:Resolver=defaultResolver):Promise<{url:URL;address:Address}>{
 if(!linkPreviewUrlSchema.safeParse(raw).success)throw new LinkPreviewBlocked();
 const url=new URL(raw),host=url.hostname.replace(/^\[|\]$/g,'').toLowerCase();
 if(isIP(host)||host==='localhost'||!host.includes('.')||/\.(local|localhost|internal|lan|home|corp|test|invalid)$/.test(host))throw new LinkPreviewBlocked();
 const addresses=await resolve(host).catch(()=>[]);
 if(!addresses.length||addresses.some(a=>!publicAddress(a.address)||isIP(a.address)!==a.family))throw new LinkPreviewBlocked();
 return {url,address:addresses[0]};
}
