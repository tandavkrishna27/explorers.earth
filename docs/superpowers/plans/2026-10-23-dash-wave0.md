```ts
/**
 * The "Recommended by" link on a place (OWN-11): a social handle or profile becomes its
 * full URL, any other absolute http(s) link is kept as typed, and anything else is
 * undefined so the form can refuse it before the contract does.
 */
export function normalizeRecommendationLink(raw?: string | null): string | undefined {
  const value = raw?.trim();
  if (!value) return undefined;
  const social = normalizeSocialProfileUrl(value);
  if (social) return social;
  try {
    const url = new URL(value);
    return url.protocol === "http:" || url.protocol === "https:" ? value : undefined;
  } catch {
    return undefined;
  }
}
```

```sql
-- OWN-01. A list's URL slug is unique only among the owner's LIVE lists in a category:
-- archiving "paris" frees "paris" for a new list. 0029 made it unique across archived
-- rows too, so an invisible archived list blocked the URL ("That URL is already used").
--
-- The full constraint guaranteed no live duplicates, so the check below cannot fire on
-- a database that reached 0064 through this chain. It stays as a loud guard for a
-- database repaired by hand: it names the offending rows instead of guessing which to keep.
-- Restoring an archived list whose slug a live list now uses fails on the new index
-- (23505 -> 409); the restore command (WP3B) turns that into "choose a new URL".
DO $$
DECLARE offenders text;
BEGIN
  SELECT string_agg(format('%s/%s/%s: %s', account_id, category, slug, ids), '; ')
    INTO offenders
    FROM (SELECT account_id, category, slug, string_agg(id::text, ',' ORDER BY id) AS ids
            FROM public.collections WHERE archived_at IS NULL
           GROUP BY account_id, category, slug HAVING count(*) > 1) AS duplicates;
  IF offenders IS NOT NULL THEN
    RAISE EXCEPTION 'OWN-01: live lists share a slug; resolve these before migrating: %', offenders;
  END IF;
END $$;

ALTER TABLE public.collections DROP CONSTRAINT collections_account_id_category_slug_key;
CREATE UNIQUE INDEX collections_live_slug_uq ON public.collections(account_id,category,slug) WHERE archived_at IS NULL;
```

```sql
-- LP-09. When an owner renames a list's URL slug, the old slug keeps working: public
-- reads match the current slug OR a slug the list was renamed from. One row per
-- (account, category, old slug). The repository keeps it unambiguous: a new list may
-- not take a slug that is still a redirect of another LIVE list, and renaming a list
-- onto a slug removes that slug's redirect. The UI for renaming is WP3B.
--
-- The foreign key uses collections' UNIQUE(id,account_id,category) (0029:33), so a
-- redirect can never point across accounts or categories. Collections are archived,
-- never deleted, at runtime; CASCADE covers only the account-deletion strip.
-- No trigger; the e2e fixture reset truncates every table.
CREATE TABLE public.collection_slug_redirects (
 account_id uuid NOT NULL,
 category text NOT NULL,
 old_slug text NOT NULL CHECK(length(old_slug) BETWEEN 1 AND 200 AND old_slug ~ '^[a-z0-9]+(-[a-z0-9]+)*$'),
 collection_id uuid NOT NULL,
 created_at timestamptz NOT NULL DEFAULT now(),
 PRIMARY KEY(account_id,category,old_slug),
 FOREIGN KEY(collection_id,account_id,category) REFERENCES public.collections(id,account_id,category) ON DELETE CASCADE
);
CREATE INDEX collection_slug_redirects_collection_idx ON public.collection_slug_redirects(collection_id);

GRANT SELECT,INSERT,UPDATE,DELETE ON public.collection_slug_redirects TO music_runtime;
```
